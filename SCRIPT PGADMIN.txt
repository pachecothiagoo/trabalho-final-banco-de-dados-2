
-- SISTEMA DE SUPRIMENTOS (ALMOXARIFADO/COMPRAS)
-- Banco de Dados 2


DROP TABLE IF EXISTS entrega_departamento  CASCADE;
DROP TABLE IF EXISTS solicitacao_interna   CASCADE;
DROP TABLE IF EXISTS recebimento_oc        CASCADE;
DROP TABLE IF EXISTS item_ordem_compra     CASCADE;
DROP TABLE IF EXISTS ordem_compra          CASCADE;
DROP TABLE IF EXISTS estoque               CASCADE;
DROP TABLE IF EXISTS produto               CASCADE;
DROP TABLE IF EXISTS departamento          CASCADE;
DROP TABLE IF EXISTS fornecedor            CASCADE;
DROP TABLE IF EXISTS funcionario           CASCADE;


CREATE TABLE funcionario (
    id_funcionario SERIAL       PRIMARY KEY,
    nome           VARCHAR(100) NOT NULL,
    cargo          VARCHAR(50)  NOT NULL
);


CREATE TABLE fornecedor (
    id_fornecedor SERIAL       PRIMARY KEY,
    razao_social  VARCHAR(120) NOT NULL,
    cnpj          VARCHAR(14)  UNIQUE NOT NULL,
    telefone      VARCHAR(20)
);


CREATE TABLE produto (
    id_produto     SERIAL       PRIMARY KEY,
    descricao      VARCHAR(120) NOT NULL,
    unidade        VARCHAR(10)  NOT NULL,
    estoque_minimo NUMERIC(10,2) DEFAULT 0
);


CREATE TABLE departamento (
    id_departamento SERIAL      PRIMARY KEY,
    nome            VARCHAR(80) NOT NULL
);


CREATE TABLE estoque (
    id_estoque       SERIAL        PRIMARY KEY,
    id_produto       INT           NOT NULL UNIQUE,
    quantidade_atual NUMERIC(10,2) NOT NULL DEFAULT 0,

    FOREIGN KEY (id_produto) REFERENCES produto(id_produto)
);



CREATE TABLE ordem_compra (
    id_oc          SERIAL      PRIMARY KEY,
    numero_oc      VARCHAR(20) UNIQUE NOT NULL,
    id_fornecedor  INT         NOT NULL,
    id_funcionario INT         NOT NULL,
    data_oc        DATE        NOT NULL,
    status         VARCHAR(20) NOT NULL DEFAULT 'PENDENTE',  -- PENDENTE, RECEBIDA, CANCELADA
    valor_total    NUMERIC(14,2) NOT NULL DEFAULT 0,

    FOREIGN KEY (id_fornecedor)  REFERENCES fornecedor(id_fornecedor),
    FOREIGN KEY (id_funcionario) REFERENCES funcionario(id_funcionario)
);



CREATE TABLE item_ordem_compra (
    id_item        SERIAL        PRIMARY KEY,
    id_oc          INT           NOT NULL,
    id_produto     INT           NOT NULL,
    quantidade     NUMERIC(10,2) NOT NULL,
    preco_unitario NUMERIC(12,2) NOT NULL,
    subtotal       NUMERIC(14,2) NOT NULL,

    FOREIGN KEY (id_oc)      REFERENCES ordem_compra(id_oc),
    FOREIGN KEY (id_produto) REFERENCES produto(id_produto)
);


CREATE TABLE recebimento_oc (
    id_recebimento      SERIAL        PRIMARY KEY,
    id_oc               INT           NOT NULL,
    id_produto          INT           NOT NULL,
    id_funcionario      INT           NOT NULL,
    quantidade_recebida NUMERIC(10,2) NOT NULL,
    preco_unitario      NUMERIC(12,2) NOT NULL,
    data_recebimento    DATE          NOT NULL,

    FOREIGN KEY (id_oc)          REFERENCES ordem_compra(id_oc),
    FOREIGN KEY (id_produto)     REFERENCES produto(id_produto),
    FOREIGN KEY (id_funcionario) REFERENCES funcionario(id_funcionario)
);


CREATE TABLE solicitacao_interna (
    id_solicitacao  SERIAL      PRIMARY KEY,
    id_departamento INT         NOT NULL,
    id_funcionario  INT         NOT NULL,
    data_solicitacao DATE       NOT NULL,
    status          VARCHAR(20) NOT NULL DEFAULT 'PENDENTE',  -- PENDENTE, ATENDIDA, NEGADA

    FOREIGN KEY (id_departamento) REFERENCES departamento(id_departamento),
    FOREIGN KEY (id_funcionario)  REFERENCES funcionario(id_funcionario)
);


CREATE TABLE item_solicitacao (
    id_item_solic  SERIAL        PRIMARY KEY,
    id_solicitacao INT           NOT NULL,
    id_produto     INT           NOT NULL,
    quantidade     NUMERIC(10,2) NOT NULL,

    FOREIGN KEY (id_solicitacao) REFERENCES solicitacao_interna(id_solicitacao),
    FOREIGN KEY (id_produto)     REFERENCES produto(id_produto)
);


CREATE TABLE entrega_departamento (
    id_entrega     SERIAL        PRIMARY KEY,
    id_solicitacao INT           NOT NULL,
    id_departamento INT          NOT NULL,
    id_produto     INT           NOT NULL,
    id_funcionario INT           NOT NULL,
    quantidade     NUMERIC(10,2) NOT NULL,
    data_entrega   DATE          NOT NULL,

    FOREIGN KEY (id_solicitacao)  REFERENCES solicitacao_interna(id_solicitacao),
    FOREIGN KEY (id_departamento) REFERENCES departamento(id_departamento),
    FOREIGN KEY (id_produto)      REFERENCES produto(id_produto),
    FOREIGN KEY (id_funcionario)  REFERENCES funcionario(id_funcionario)
);

CREATE OR REPLACE PROCEDURE sp_inserir_produto(
    p_descricao VARCHAR(120),
    p_unidade VARCHAR(10),
    p_estoque_minimo NUMERIC(10,2),
    OUT p_id_gerado INT
)
LANGUAGE plpgsql AS $$
BEGIN

    INSERT INTO produto (descricao, unidade, estoque_minimo)
    VALUES (p_descricao, p_unidade, p_estoque_minimo)
    RETURNING id_produto INTO p_id_gerado;


    INSERT INTO estoque (id_produto, quantidade_atual)
    VALUES (p_id_gerado, 0.00);
END;
$$;


CREATE OR REPLACE PROCEDURE sp_registrar_recebimento(
    p_id_oc INT,
    p_id_produto INT,
    p_id_funcionario INT,
    p_quantidade NUMERIC(10,2),
    p_preco_unitario NUMERIC(12,2)
)
LANGUAGE plpgsql AS $$
BEGIN

    INSERT INTO recebimento_oc (id_oc, id_produto, id_funcionario, quantidade_recebida, preco_unitario, data_recebimento)
    VALUES (p_id_oc, p_id_produto, p_id_funcionario, p_quantidade, p_preco_unitario, CURRENT_DATE);


    UPDATE estoque 
    SET quantidade_atual = quantidade_atual + p_quantidade
    WHERE id_produto = p_id_produto;
END;
$$;

CREATE OR REPLACE PROCEDURE sp_registrar_entrega(
    p_id_solicitacao INT,
    p_id_departamento INT,
    p_id_produto INT,
    p_id_funcionario INT,
    p_quantidade NUMERIC(10,2)
)
LANGUAGE plpgsql AS $$
DECLARE
    v_estoque_disponivel NUMERIC(10,2);
BEGIN

    SELECT quantidade_atual INTO v_estoque_disponivel 
    FROM estoque 
    WHERE id_produto = p_id_produto;

    IF v_estoque_disponivel < p_quantidade THEN
        RAISE EXCEPTION 'Saldo insuficiente em estoque para o produto %.', p_id_produto;
    END IF;

    INSERT INTO entrega_departamento (id_solicitacao, id_departamento, id_produto, id_funcionario, quantidade, data_entrega)
    VALUES (p_id_solicitacao, p_id_departamento, p_id_produto, p_id_funcionario, p_quantidade, CURRENT_DATE);

    UPDATE estoque 
    SET quantidade_atual = quantidade_atual - p_quantidade
    WHERE id_produto = p_id_produto;
END;
$$;


CREATE OR REPLACE FUNCTION fn_impedir_exclusao_produto()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
    IF (SELECT quantidade_atual FROM estoque WHERE id_produto = OLD.id_produto) > 0 THEN
        RAISE EXCEPTION 'Não é permitido excluir um produto que ainda possui saldo em estoque.';
    END IF;
    
    DELETE FROM estoque WHERE id_produto = OLD.id_produto;
    
    RETURN OLD;
END;
$$;

CREATE TRIGGER tg_antes_deletar_produto
BEFORE DELETE ON produto
FOR EACH ROW
EXECUTE FUNCTION fn_impedir_exclusao_produto();


CREATE OR REPLACE VIEW vw_dados_produtos AS
SELECT 
    p.id_produto,
    p.descricao,
    p.unidade,
    p.estoque_minimo,
    COALESCE(e.quantidade_atual, 0) AS estoque_atual,
    CASE 
        WHEN COALESCE(e.quantidade_atual, 0) <= p.estoque_minimo THEN 'COMPRAR'
        ELSE 'OK'
    END AS status_estoque
FROM produto p
LEFT JOIN estoque e ON p.id_produto = e.id_produto;

CREATE OR REPLACE VIEW vw_consumo_departamentos AS
SELECT 
    d.nome AS departamento,
    p.descricao AS produto,
    SUM(ed.quantidade) AS total_entregue
FROM entrega_departamento ed
JOIN departamento d ON ed.id_departamento = d.id_departamento
JOIN produto p ON ed.id_produto = p.id_produto
GROUP BY d.nome, p.descricao;

CREATE OR REPLACE VIEW vw_compras_pendentes AS
SELECT 
    oc.numero_oc,
    f.razao_social AS fornecedor,
    oc.data_oc,
    oc.valor_total
FROM ordem_compra oc
JOIN fornecedor f ON oc.id_fornecedor = f.id_fornecedor
WHERE oc.status = 'PENDENTE';
